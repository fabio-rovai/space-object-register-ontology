import json, sys, pathlib, collections
sys.path.insert(0, str(pathlib.Path(__file__).resolve().parent))
import rdflib
from reconcile import load_celestrak, load_gcat, GONE, LEFT_EARTH, ERROR, LOST

ROOT = pathlib.Path(__file__).resolve().parent.parent
NS = "https://gov.tesseract.academy/def/space#"

def verify_disposition_disagreement():
    """Independent verification of DispositionDisagreement count.
    
    Uses explicit iteration with separate classification steps
    instead of a set comprehension, providing an independent code path
    for cross-checking.
    """
    ct = load_celestrak()
    sat, _, _ = load_gcat("gcat_satcat.tsv")
    
    ctdec = {n for n in ct if ct[n]["DECAY_DATE"].strip()}
    st = {n: {p.get("Status", "").strip() for p in v} for n, v in sat.items()}
    
    count = 0
    for norad_id in sat:
        if norad_id not in ct:
            continue
        
        # Classify disposition in GCAT
        gcat_has_gone = bool(st[norad_id] & GONE)
        gcat_has_left_earth = bool(st[norad_id] & LEFT_EARTH)
        
        # Classify disposition in CelesTrak
        ct_has_decay = norad_id in ctdec
        
        # Check for disagreement
        if gcat_has_gone != ct_has_decay:
            # Exception: LEFT_EARTH without GONE but with decay date is acceptable
            if ct_has_decay and gcat_has_left_earth and not gcat_has_gone:
                continue
            count += 1
    
    return count

def python_side():
    ct = load_celestrak()
    sat, _, nona = load_gcat("gcat_satcat.tsv")
    aux, _, _ = load_gcat("gcat_auxcat.tsv")
    fto, _, _ = load_gcat("gcat_ftocat.tsv")
    allg = set(sat) | set(aux) | set(fto)
    st = {n: {p.get("Status", "").strip() for p in v} for n, v in sat.items()}
    ctdec = {n for n in ct if ct[n]["DECAY_DATE"].strip()}
    phantom = {n for n in sat if (st[n] & ERROR) and n in ct}
    
    # Compute DispositionDisagreement using original method
    disposition_disagreement = len({
        n for n in sat if n in ct
        and (bool(st[n] & GONE) != (n in ctdec))
        and not ((n in ctdec) and (st[n] & LEFT_EARTH) and not (st[n] & GONE))})
    
    # Verify using independent method
    disposition_disagreement_verify = verify_disposition_disagreement()
    
    # Cross-check
    if disposition_disagreement != disposition_disagreement_verify:
        raise AssertionError(
            f"DispositionDisagreement mismatch: {disposition_disagreement} vs {disposition_disagreement_verify}")
    
    return {
        "PhantomEntry": len(phantom),
        "PhantomEntryOnOrbit": len({n for n in phantom if n not in ctdec}),
        "UndisclosedTrackingLoss": len({n for n in sat if (st[n] & LOST) and n in ct and n not in ctdec}),
        "DispositionDisagreement": disposition_disagreement,
        "CoverageGap": len(set(ct) - allg),
        "UnnumberedObject": len(nona),
        "IdentifierCollision": len({n for n in sat if len({p.get("JCAT","")[:6] for p in sat[n]}) > 1}),
    }
